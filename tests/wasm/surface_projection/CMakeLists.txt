cmake_minimum_required(VERSION 3.16)
project(TerraSurfaceProjectionContract LANGUAGES C CXX)
get_filename_component(TERRA_SDK_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_subdirectory("${TERRA_SDK_SOURCE_ROOT}/adapters/wasm" sdk-wasm)
target_sources(terra_sdk_wasm PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../surface_projection_contract.cpp")
target_sources(terra_sdk_wasm PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../surface_model_contract.cpp")
get_target_property(contract_options terra_sdk_wasm LINK_OPTIONS)
set(expanded_options)
foreach(option IN LISTS contract_options)
    if(option MATCHES "^-sEXPORTED_FUNCTIONS=")
        string(APPEND option ",_test_triangle_project,_test_projection_create,_test_projection_step,_test_projection_block,_test_projection_admit,_test_projection_read,_test_projection_summary,_test_projection_cancel,_test_projection_destroy,_test_projection_handles")
        string(APPEND option ",_test_model_create,_test_model_step,_test_model_admit,_test_model_block,_test_model_read,_test_model_same_layout,_test_model_measure,_test_model_measure_step,_test_model_measure_read,_test_model_summary,_test_model_input,_test_model_measure_cancel,_test_model_destroy,_test_model_handles")
    endif()
    list(APPEND expanded_options "${option}")
endforeach()
set_target_properties(terra_sdk_wasm PROPERTIES LINK_OPTIONS "${expanded_options}"
    OUTPUT_NAME surface_projection_contract RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
