cmake_minimum_required(VERSION 3.16)
project(TerraRegionCompactionContract LANGUAGES C CXX)
get_filename_component(TERRA_SDK_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_subdirectory("${TERRA_SDK_SOURCE_ROOT}/adapters/wasm" sdk-wasm)
target_sources(terra_sdk_wasm PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../region_compaction_contract.cpp")
get_target_property(contract_options terra_sdk_wasm LINK_OPTIONS)
set(contract_expanded)
foreach(option IN LISTS contract_options)
    if(option MATCHES "^-sEXPORTED_FUNCTIONS=")
        string(APPEND option ",_test_region_quote,_test_region_create,_test_region_step,_test_region_admit,_test_region_admit_reserved,_test_region_unblock,_test_region_read,_test_region_cancel,_test_region_destroy,_test_region_handles")
    endif()
    list(APPEND contract_expanded "${option}")
endforeach()
set_target_properties(terra_sdk_wasm PROPERTIES LINK_OPTIONS "${contract_expanded}"
    OUTPUT_NAME region_compaction_contract RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
