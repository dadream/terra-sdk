cmake_minimum_required(VERSION 3.16)
project(TerraMeshResidencyDiagnostic LANGUAGES C CXX)
get_filename_component(TERRA_SDK_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_subdirectory("${TERRA_SDK_SOURCE_ROOT}/adapters/wasm" sdk-wasm)
get_target_property(diagnostic_sources terra_sdk_wasm SOURCES)
list(FILTER diagnostic_sources EXCLUDE REGEX "/sdk/src/c_api/terra\\.cpp$")
set_property(TARGET terra_sdk_wasm PROPERTY SOURCES "${diagnostic_sources}")
target_sources(terra_sdk_wasm PRIVATE "${CMAKE_CURRENT_LIST_DIR}/snapshot.cpp")
get_target_property(diagnostic_options terra_sdk_wasm LINK_OPTIONS)
set(diagnostic_expanded)
foreach(option IN LISTS diagnostic_options)
    if(option MATCHES "^-sEXPORTED_FUNCTIONS=")
        string(APPEND option ",_test_mesh_residency")
    endif()
    list(APPEND diagnostic_expanded "${option}")
endforeach()
set_target_properties(terra_sdk_wasm PROPERTIES LINK_OPTIONS "${diagnostic_expanded}"
    OUTPUT_NAME mesh_residency_diagnostic RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
